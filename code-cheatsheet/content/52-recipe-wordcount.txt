=== python
text = "the cat and the dog and the bird"

counts = {}
for w in text.split():
    counts[w] = counts.get(w, 0) + 1

for w, n in sorted(counts.items(), key=lambda kv: (-kv[1], kv[0])):
    print(f"{w}: {n}")
=== javascript
const text = "the cat and the dog and the bird";

const counts = {};
for (const w of text.split(" ")) {
  counts[w] = (counts[w] ?? 0) + 1;
}

const sorted = Object.entries(counts).sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]));
for (const [w, n] of sorted) {
  console.log(`${w}: ${n}`);
}
=== c
#include <stdio.h>
#include <string.h>

int main(void) {
    char *words[8] = {"the", "cat", "and", "the", "dog", "and", "the", "bird"};
    for (int i = 0; i < 8; i++) {
        int seen = 0;
        for (int j = 0; j < i; j++) {
            if (strcmp(words[i], words[j]) == 0) {
                seen = 1;
            }
        }
        if (seen) {
            continue;
        }
        int count = 0;
        for (int j = 0; j < 8; j++) {
            if (strcmp(words[i], words[j]) == 0) {
                count++;
            }
        }
        printf("%s: %d\n", words[i], count);
    }
    return 0;
}
=== kotlin
fun main() {
    val text = "the cat and the dog and the bird"

    val counts = text.split(" ").groupingBy { it }.eachCount()
    val sorted = counts.entries.sortedWith(compareBy({ -it.value }, { it.key }))
    for ((w, n) in sorted) {
        println("$w: $n")
    }
}

def is_prime(n):
    if n < 2:
        return False
    i = 2
    while i * i <= n:
        if n % i == 0:
            return False
        i += 1
    return True


primes = []
for n in range(2, 31):
    if is_prime(n):
        primes.append(n)
print(primes)

function isPrime(n) {
  if (n < 2) return false;
  for (let i = 2; i * i <= n; i++) {
    if (n % i === 0) return false;
  }
  return true;
}

const primes = [];
for (let n = 2; n <= 30; n++) {
  if (isPrime(n)) primes.push(n);
}
console.log(primes);

#include <stdio.h>

int is_prime(int n) {
    if (n < 2) {
        return 0;
    }
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) {
            return 0;
        }
    }
    return 1;
}

int main(void) {
    int first = 1;
    printf("[");
    for (int n = 2; n <= 30; n++) {
        if (is_prime(n)) {
            if (!first) {
                printf(", ");
            }
            printf("%d", n);
            first = 0;
        }
    }
    printf("]\n");
    return 0;
}

fun isPrime(n: Int): Boolean {
    if (n < 2) return false
    var i = 2
    while (i * i <= n) {
        if (n % i == 0) return false
        i++
    }
    return true
}

fun main() {
    val primes = (2..30).filter { isPrime(it) }
    println(primes)
}

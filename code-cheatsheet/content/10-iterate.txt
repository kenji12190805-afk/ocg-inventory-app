=== python
fruits = ["りんご", "みかん", "ぶどう"]
for fruit in fruits:
    print(fruit)
for i, fruit in enumerate(fruits, start=1):
    print(i, fruit)

=== c
#include <stdio.h>

int main(void) {
    char *fruits[] = {"apple", "orange", "grape"};
    int n = sizeof(fruits) / sizeof(fruits[0]);
    for (int i = 0; i < n; i++) {
        printf("%d %s\n", i + 1, fruits[i]);
    }
    return 0;
}

    }

    }
}

=== c
#include <stdio.h>

int main(void) {
    int nums[5] = {1, 2, 3, 4, 5};
    int sum = 0;
    for (int i = 0; i < 5; i++) {
        printf("double: %d\n", nums[i] * 2);
    }
    for (int i = 0; i < 5; i++) {
        if (nums[i] % 2 == 0) {
            printf("even: %d\n", nums[i]);
        }
        sum += nums[i];
    }
    printf("sum: %d\n", sum);
    return 0;
}

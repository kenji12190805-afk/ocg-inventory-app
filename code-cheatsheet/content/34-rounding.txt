=== c
#include <stdio.h>
#include <stdlib.h>
#include <math.h>

int main(void) {
    printf("%.2f\n", 3.14159);
    printf("%.1f\n", sqrt(16.0));
    printf("%.0f\n", pow(2.0, 10.0));
    printf("%d\n", abs(-5));
    printf("%.0f %.0f\n", floor(3.7), ceil(3.2));
    return 0;
}

=== c
#include <stdio.h>

int main(void) {
    int nums[6] = {1, 2, 2, 3, 3, 3};
    int uniq[6];
    int count = 0;
    for (int i = 0; i < 6; i++) {
        int found = 0;
        for (int j = 0; j < count; j++) {
            if (uniq[j] == nums[i]) {
                found = 1;
            }
        }
        if (!found) {
            uniq[count] = nums[i];
            count++;
        }
    }
    printf("count: %d\n", count);
    for (int j = 0; j < count; j++) {
        printf("%d\n", uniq[j]);
    }
    return 0;
}

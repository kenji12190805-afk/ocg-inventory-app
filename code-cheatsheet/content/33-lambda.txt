=== python
def double(n):
    return n * 2


def apply_to(f, x):
    return f(x)


print(apply_to(double, 5))
print(apply_to(lambda n: n + 1, 5))
print(list(map(lambda n: n * n, [1, 2, 3])))

=== c
#include <stdio.h>

int twice(int x) {
    return x * 2;
}

int square(int x) {
    return x * x;
}

int apply_to(int (*f)(int), int v) {
    return f(v);
}

int main(void) {
    printf("%d\n", apply_to(twice, 5));
    printf("%d\n", apply_to(square, 5));
    return 0;
}

#include <stdio.h>
#include <string.h>

int main(void) {
    char s[] = "Hello, World";
    char part[8];
    strncpy(part, s, 5);
    part[5] = '\0';
    printf("%s\n", part);
    printf("%s\n", &s[7]);
    return 0;
}

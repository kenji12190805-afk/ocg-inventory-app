=== c
#include <stdio.h>

int main(void) {
    for (int c = 0; c <= 40; c += 10) {
        double f = c * 9.0 / 5 + 32;
        printf("%d C = %.1f F\n", c, f);
    }
    return 0;
}
=== kotlin
fun main() {
    for (c in 0..40 step 10) {
        val f = c * 9.0 / 5 + 32
        println("$c°C = ${"%.1f".format(f)}°F")
    }
}

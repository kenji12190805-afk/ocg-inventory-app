=== python
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age

    def intro(self):
        print(f"{self.name}です。{self.age}歳です。")

p = Person("ケンジ", 30)
p.intro()

class Person {
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }
  intro() {
    console.log(`${this.name}です。${this.age}歳です。`);
  }
}

const p = new Person("ケンジ", 30);
p.intro();

=== c
#include <stdio.h>
#include <string.h>

struct Person {
    char name[32];
    int age;
};

void intro(struct Person p) {
    printf("I am %s. %d years old.\n", p.name, p.age);
}

int main(void) {
    struct Person p;
    strcpy(p.name, "Kenji");
    p.age = 30;
    intro(p);
    return 0;
}

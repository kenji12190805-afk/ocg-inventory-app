=== c
#include <stdio.h>

int main(void) {
    int scores[3] = {80, 65, 90};
    int count = sizeof(scores) / sizeof(scores[0]);
    printf("%d\n", scores[0]);
    printf("%d\n", scores[count - 1]);
    printf("%d\n", count);
    return 0;
}

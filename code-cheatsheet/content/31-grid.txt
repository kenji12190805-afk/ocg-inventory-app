=== c
#include <stdio.h>

int main(void) {
    int grid[2][3] = {{1, 2, 3}, {4, 5, 6}};
    printf("%d\n", grid[1][2]);
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 3; j++) {
            if (j > 0) {
                printf(" ");
            }
            printf("%d", grid[i][j]);
        }
        printf("\n");
    }
    return 0;
}

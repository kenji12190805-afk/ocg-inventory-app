=== python
fib = []
a, b = 0, 1
for _ in range(10):
    fib.append(a)
    a, b = b, a + b
print(fib)

#include <stdio.h>

int main(void) {
    int a = 0;
    int b = 1;
    printf("[");
    for (int i = 0; i < 10; i++) {
        if (i > 0) {
            printf(", ");
        }
        printf("%d", a);
        int next = a + b;
        a = b;
        b = next;
    }
    printf("]\n");
    return 0;
}

fun main() {
    val fib = mutableListOf<Int>()
    var a = 0
    var b = 1
    repeat(10) {
        fib.add(a)
        val next = a + b
        a = b
        b = next
    }
    println(fib)
}

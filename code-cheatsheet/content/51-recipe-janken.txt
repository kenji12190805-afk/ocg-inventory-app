=== python
hands = ["グー", "チョキ", "パー"]


def judge(a, b):
    if a == b:
        return "あいこ"
    if (a - b + 3) % 3 == 2:
        return "勝ち"
    return "負け"


for a, b in [(0, 0), (0, 1), (0, 2), (1, 2)]:
    print(f"{hands[a]} vs {hands[b]}: {judge(a, b)}")

const hands = ["グー", "チョキ", "パー"];

function judge(a, b) {
  if (a === b) return "あいこ";
  if ((a - b + 3) % 3 === 2) return "勝ち";
  return "負け";
}

for (const [a, b] of [[0, 0], [0, 1], [0, 2], [1, 2]]) {
  console.log(`${hands[a]} vs ${hands[b]}: ${judge(a, b)}`);

=== c
#include <stdio.h>

char *judge(int a, int b) {
    if (a == b) {
        return "draw";
    }
    if ((a - b + 3) % 3 == 2) {
        return "win";
    }
    return "lose";
}

int main(void) {
    char *hands[3] = {"Rock", "Scissors", "Paper"};
    int pairs[4][2] = {{0, 0}, {0, 1}, {0, 2}, {1, 2}};
    for (int i = 0; i < 4; i++) {
        int a = pairs[i][0];
        int b = pairs[i][1];
        printf("%s vs %s: %s\n", hands[a], hands[b], judge(a, b));
    }
    return 0;
}

val hands = listOf("グー", "チョキ", "パー")

fun judge(a: Int, b: Int): String {
    if (a == b) return "あいこ"
    if ((a - b + 3) % 3 == 2) return "勝ち"
    return "負け"
}

fun main() {
    for ((a, b) in listOf(Pair(0, 0), Pair(0, 1), Pair(0, 2), Pair(1, 2))) {
        println("${hands[a]} vs ${hands[b]}: ${judge(a, b)}")
    }
}

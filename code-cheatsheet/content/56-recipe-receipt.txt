items = [("りんご", 120, 3), ("みかん", 80, 5), ("ぶどう", 450, 1)]

subtotal = 0
for name, price, qty in items:
    line = price * qty
    subtotal += line
    print(f"{name} {price}円 x {qty} = {line}円")

tax = subtotal // 10
print(f"小計: {subtotal}円")
print(f"税(10%): {tax}円")

=== javascript
const items = [["りんご", 120, 3], ["みかん", 80, 5], ["ぶどう", 450, 1]];

let subtotal = 0;
for (const [name, price, qty] of items) {
  const line = price * qty;
  subtotal += line;
  console.log(`${name} ${price}円 x ${qty} = ${line}円`);
}

const tax = Math.floor(subtotal / 10);
console.log(`小計: ${subtotal}円`);
console.log(`税(10%): ${tax}円`);
console.log(`合計: ${subtotal + tax}円`);
=== c
#include <stdio.h>

int main(void) {
    char *names[3] = {"apple", "orange", "grape"};
    int prices[3] = {120, 80, 450};
    int qtys[3] = {3, 5, 1};
    int subtotal = 0;
    for (int i = 0; i < 3; i++) {
        int line = prices[i] * qtys[i];
        subtotal += line;
        printf("%s %d yen x %d = %d yen\n", names[i], prices[i], qtys[i], line);
    }
    int tax = subtotal / 10;
    printf("subtotal: %d yen\n", subtotal);
    printf("tax(10%%): %d yen\n", tax);
    printf("total: %d yen\n", subtotal + tax);
    return 0;
}

=== python
class Animal:
    def __init__(self, name):
        self.name = name

    def speak(self):
        print(f"{self.name}は鳴く")


class Dog(Animal):
    def __init__(self, name, tricks):
        super().__init__(name)
        self.tricks = tricks

    def speak(self):
        print(f"{self.name}はワンと鳴く(芸: {self.tricks})")


Animal("動物").speak()
Dog("ポチ", 2).speak()

=== javascript
class Animal {
  constructor(name) {
    this.name = name;
  }
  speak() {
    console.log(`${this.name}は鳴く`);
  }
}

class Dog extends Animal {
  constructor(name, tricks) {
    super(name);
    this.tricks = tricks;
  }
  speak() {
    console.log(`${this.name}はワンと鳴く(芸: ${this.tricks})`);
  }
}

new Animal("動物").speak();
new Dog("ポチ", 2).speak();

=== c
#include <stdio.h>
#include <string.h>

struct Animal {
    char name[16];
};

struct Dog {
    struct Animal base;
    int tricks;
};

int main(void) {
    struct Dog d;
    strcpy(d.base.name, "Pochi");
    d.tricks = 2;
    printf("%s knows %d tricks\n", d.base.name, d.tricks);
    return 0;
}

=== kotlin
open class Animal(val name: String) {
    open fun speak() {
        println("${name}は鳴く")
    }
}

class Dog(name: String, val tricks: Int) : Animal(name) {
    override fun speak() {
        println("${name}はワンと鳴く(芸: $tricks)")
    }
}

fun main() {
    Animal("動物").speak()
    Dog("ポチ", 2).speak()
}

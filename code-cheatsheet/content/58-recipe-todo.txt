=== c
#include <stdio.h>
#include <string.h>

struct Todo {
    char title[32];
    int done;
};

int main(void) {
    struct Todo todos[3];
    strcpy(todos[0].title, "buy milk");
    todos[0].done = 0;
    strcpy(todos[1].title, "homework");
    todos[1].done = 0;
    strcpy(todos[2].title, "clean up");
    todos[2].done = 0;

    todos[1].done = 1;

    for (int i = 0; i < 3; i++) {
        char mark = ' ';
        if (todos[i].done) {
            mark = 'x';
        }
        printf("%d. %s [%c]\n", i + 1, todos[i].title, mark);
    }
    return 0;
}
=== kotlin
class Todo(val title: String, var done: Boolean)

fun main() {
    val todos = listOf(
        Todo("牛乳を買う", false),
        Todo("宿題をする", false),
        Todo("掃除する", false),
    )

    todos[1].done = true

    for ((i, t) in todos.withIndex()) {
        val mark = if (t.done) "x" else " "
        println("${i + 1}. ${t.title} [$mark]")
    }
}

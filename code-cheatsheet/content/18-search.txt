=== c
#include <stdio.h>
#include <string.h>

int main(void) {
    char s[] = "apple,banana,cherry";
    if (strstr(s, "banana") != NULL) {
        printf("found banana\n");
    }
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == ',') {
            s[i] = ' ';
        }
    }
    printf("%s\n", s);
    return 0;
}

=== c
#include <stdio.h>

int main(void) {
    char *words[] = {"apple", "banana", "cherry"};
    for (int i = 0; i < 3; i++) {
        if (i > 0) {
            printf(", ");
        }
        printf("%s", words[i]);
    }
    printf("\n");
    for (int i = 0; i < 3; i++) {
        printf("ab");
    }
    printf("\n");
    return 0;
}

scores = [80, 65, 92, 48, 75]

for s in scores:
    result = "合格" if s >= 60 else "不合格"
    print(f"{s}点: {result}")

print("合計:", sum(scores))
print("平均:", sum(scores) / len(scores))

const scores = [80, 65, 92, 48, 75];

for (const s of scores) {
  const result = s >= 60 ? "合格" : "不合格";
  console.log(`${s}点: ${result}`);
}

const total = scores.reduce((a, b) => a + b, 0);
console.log("合計:", total);
console.log("平均:", total / scores.length);
console.log("最高:", Math.max(...scores));
=== c
#include <stdio.h>

int main(void) {
    int scores[5] = {80, 65, 92, 48, 75};
    int sum = 0;
    int max = scores[0];
    for (int i = 0; i < 5; i++) {
        if (scores[i] >= 60) {
            printf("%d: pass\n", scores[i]);
        } else {
            printf("%d: fail\n", scores[i]);
        }
        sum += scores[i];
        if (scores[i] > max) {
            max = scores[i];
        }
    }
    printf("sum: %d\n", sum);
    printf("average: %.1f\n", (double)sum / 5);
    printf("max: %d\n", max);
    return 0;
}

fun main() {
    val scores = listOf(80, 65, 92, 48, 75)

    for (s in scores) {
        val result = if (s >= 60) "合格" else "不合格"
        println("${s}点: $result")
    }

    println("合計: ${scores.sum()}")
    println("平均: ${scores.average()}")
    println("最高: ${scores.max()}")
}

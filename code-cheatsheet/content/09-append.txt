=== c
#include <stdio.h>

int main(void) {
    int nums[10] = {1, 2, 3};
    int count = 3;
    nums[count] = 4;
    count++;
    for (int i = 0; i < count; i++) {
        printf("%d\n", nums[i]);
    }
    return 0;
}

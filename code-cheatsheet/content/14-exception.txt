=== c
#include <stdio.h>

int divide(int a, int b, int *result) {
    if (b == 0) {
        return -1;
    }
    *result = a / b;
    return 0;
}

int main(void) {
    int r;
    if (divide(10, 0, &r) != 0) {
        printf("cannot divide by zero\n");
    } else {
        printf("%d\n", r);
    }
    return 0;
}

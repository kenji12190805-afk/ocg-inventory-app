=== c
#include <stdio.h>
#include <string.h>

struct Entry {
    char name[16];
    int age;
};

int main(void) {
    struct Entry ages[3];
    strcpy(ages[0].name, "Kenji");
    ages[0].age = 30;
    strcpy(ages[1].name, "Hana");
    ages[1].age = 25;
    strcpy(ages[2].name, "Taro");
    ages[2].age = 20;
    for (int i = 0; i < 3; i++) {
        if (strcmp(ages[i].name, "Hana") == 0) {
            printf("%s is %d\n", ages[i].name, ages[i].age);
        }
    }
    return 0;
}

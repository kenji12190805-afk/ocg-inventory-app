=== c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main(void) {
    char s[32] = "Hello";
    printf("%d\n", (int)strlen(s));
    strcat(s, " World");
    printf("%s\n", s);
    for (int i = 0; s[i] != '\0'; i++) {
        s[i] = toupper(s[i]);
    }
    printf("%s\n", s);
    return 0;
}

=== c
#include <stdio.h>

int main(void) {
    int a[5] = {3, 9, 1, 7, 5};
    int max = a[0];
    int min = a[0];
    int sum = 0;
    for (int i = 0; i < 5; i++) {
        if (a[i] > max) {
            max = a[i];
        }
        if (a[i] < min) {
            min = a[i];
        }
        sum += a[i];
    }
    printf("%d %d\n", max, min);
    printf("%.1f\n", (double)sum / 5);
    return 0;
}

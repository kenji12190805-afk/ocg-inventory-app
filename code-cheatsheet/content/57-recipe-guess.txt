secret = 37
low, high = 1, 100
tries = 0

while True:
    mid = (low + high) // 2
    tries += 1
    if mid == secret:
        print(f"{mid}: 正解! ({tries}回)")
        break
    elif mid > secret:
        print(f"{mid}: 大きい")
        high = mid - 1
    else:
        print(f"{mid}: 小さい")

=== javascript
const secret = 37;
let low = 1;
let high = 100;
let tries = 0;

while (true) {
  const mid = Math.floor((low + high) / 2);
  tries++;
  if (mid === secret) {
    console.log(`${mid}: 正解! (${tries}回)`);
    break;
  } else if (mid > secret) {
    console.log(`${mid}: 大きい`);
    high = mid - 1;
  } else {
    console.log(`${mid}: 小さい`);
    low = mid + 1;
  }
}
=== c
#include <stdio.h>

int main(void) {
    int secret = 37;
    int low = 1;
    int high = 100;
    int tries = 0;
    while (1) {
        int mid = (low + high) / 2;
        tries++;
        if (mid == secret) {
            printf("%d: correct! (%d tries)\n", mid, tries);
            break;
        } else if (mid > secret) {
            printf("%d: too big\n", mid);
            high = mid - 1;
        } else {
            printf("%d: too small\n", mid);
            low = mid + 1;
        }
    }
    return 0;
}
